CREATE TABLE VEHICULO_EXAMEN (
    placa VARCHAR(20) PRIMARY KEY,
    tipo VARCHAR2(20) CHECK (tipo IN ('OFICIAL', 'RESIDENTE', 'NO_RESIDENTE')),
    entrada_actual DATE DEFAULT NULL,
    minutos_acumulados INT DEFAULT 0
);
